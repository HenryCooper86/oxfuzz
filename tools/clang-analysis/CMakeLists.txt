cmake_minimum_required(VERSION 3.20)
project(oxfuzz_clang_analysis LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 18)
  message(FATAL_ERROR "Production helper requires LLVM 18")
endif()
add_executable(oxfuzz-clang-analyze main.cpp exploration.cpp)
target_compile_features(oxfuzz-clang-analyze PRIVATE cxx_std_17)
target_include_directories(oxfuzz-clang-analyze SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_link_libraries(oxfuzz-clang-analyze PRIVATE clang-cpp LLVM)
install(TARGETS oxfuzz-clang-analyze RUNTIME DESTINATION bin)
